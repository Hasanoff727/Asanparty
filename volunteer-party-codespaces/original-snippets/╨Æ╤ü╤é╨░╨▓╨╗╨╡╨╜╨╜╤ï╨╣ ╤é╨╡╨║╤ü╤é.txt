"""Transactional email via Emergent's managed Resend integration (RESEND_EMAIL_PLAYBOOK).

Fire-and-forget by design: a registration must succeed even when email is down.
The structural gate `_assert_safe_email` runs on EVERY send path — never weaken it.
G4: recipients come from server-side records; bodies are fixed server-side templates.
"""

import asyncio
import ipaddress
import logging
import os
import re
from html import escape
from html.parser import HTMLParser
from urllib.parse import urlparse

import httpx
from dotenv import load_dotenv

from lib.db import db

load_dotenv()
logger = logging.getLogger(__name__)

# Emergent managed email proxy. This is a CONSTANT — never read it from os.environ.
EMAIL_BASE_URL = "https://integrations.emergentagent.com"
EMAIL_KEY = os.environ.get("EMERGENT_EMAIL_KEY", "")
EMAIL_FROM_NAME = os.environ.get("EMAIL_FROM_NAME", "")  # this app's OWN brand (G1)
APP_URL = os.environ.get("APP_URL", "").rstrip("/")
EMAIL_ENABLED = bool(EMAIL_KEY and EMAIL_FROM_NAME)

_SHORTENERS = ("bit.ly", "tinyurl.com", "t.co", "is.gd", "cutt.ly", "goo.gl", "rebrand.ly")
_CRED_ASK = ("reply with your password", "reply with the code", "send your password", "cvv",
             "send us your password", "enter your password below", "confirm your card number",
             "your full card number", "seed phrase", "recovery phrase", "verify your card",
             "social security number", "confirm your bank details")  # G2 ask-back phrasing
_HOSTISH = re.compile(r"\b(?:https?://)?((?:[a-z0-9-]+\.)+[a-z]{2,})", re.I)  # host-looking text


def _host_ok(host: str) -> bool:
    """Reject empty, punycode, IP-literal (v4 AND v6), and shortener hosts."""
    if not host or "xn--" in host:
        return False
    try:
        ipaddress.ip_address(host)  # any IPv4 or IPv6 literal (incl. [::1])
        return False
    except ValueError:
        pass
    return not any(host == s or host.endswith("." + s) for s in _SHORTENERS)


def _same_site(shown: str, real: str) -> bool:
    """Share a registrable domain by label-boundary suffix."""
    return shown == real or real.endswith("." + shown) or shown.endswith("." + real)


class _EmailScan(HTMLParser):
    """Real parsing, not regex: collects each anchor's DECODED visible text."""

    def __init__(self):
        super().__init__()
        self.tags, self.urls, self.anchors = set(), [], []
        self._href, self._text = None, []

    def handle_starttag(self, tag, attrs):
        self.tags.add(tag.lower())
        self.urls += [v for k, v in attrs if k.lower() in ("href", "src") and v]
        if tag.lower() == "a":
            self._href = dict((k.lower(), v) for k, v in attrs).get("href")
            self._text = []

    def handle_data(self, data):
        if self._href is not None:
            self._text.append(data)

    def handle_endtag(self, tag):
        if tag.lower() == "a" and self._href is not None:
            self.anchors.append((self._href, "".join(self._text)))
            self._href, self._text = None, []


def _assert_safe_email(subject: str, html: str) -> None:
    """Best-effort structural check for G2 + G3 (defense in depth, NOT G1)."""
    scan = _EmailScan()
    scan.feed(html)
    if scan.tags & {"form", "input", "textarea", "select"}:
        raise ValueError("No forms or input fields in email (G2)")
    body = f"{subject}\n{html}".lower()
    for p in _CRED_ASK:
        if p in body:
            raise ValueError(f"Email asks the recipient for credentials: {p!r} (G2)")
    for url in scan.urls:  # every href AND src
        low = url.strip().lower()
        if low.startswith(("mailto:", "tel:", "cid:", "#")):
            continue
        if not low.startswith("https://"):  # also kills javascript:, data:, relative
            raise ValueError(f"Email links/assets must be absolute https: {url!r} (G3)")
        host = urlparse(low).hostname or ""
        if not _host_ok(host) or urlparse(low).username is not None:
            raise ValueError(f"Shortened, numeric-host or credential-bearing URL: {url!r} (G3)")
    for href, text in scan.anchors:  # decoded visible text vs the real host
        real = urlparse(href.strip().lower()).hostname or ""
        if not real:
            continue
        for m in _HOSTISH.finditer(text):  # EVERY host-like token, not just the first
            if not _same_site(m.group(1).lower(), real):
                raise ValueError(f"Anchor text {m.group(1)!r} ≠ real link host {real!r} (G3)")


async def _send_raw(*, to: str, subject: str, html: str) -> str | None:
    if not EMAIL_ENABLED:
        logger.warning("Email skipped: EMERGENT_EMAIL_KEY/EMAIL_FROM_NAME not configured")
        return None
    _assert_safe_email(subject, html)  # G2-G3 gate — every send path, never skipped
    payload = {"to": [to], "subject": subject, "html": html, "from_name": EMAIL_FROM_NAME}
    try:
        async with httpx.AsyncClient(timeout=30) as client:
            resp = await client.post(
                f"{EMAIL_BASE_URL}/api/v1/email/send",
                headers={"X-Email-Key": EMAIL_KEY},
                json=payload,
            )
        resp.raise_for_status()
        return resp.json().get("id")
    except httpx.HTTPStatusError as exc:
        try:
            code = exc.response.json().get("code", "")
        except ValueError:
            code = ""
        if code in ("insufficient_credits", "integration_disabled", "daily_limit_reached"):
            # Billing state on the app owner's account — log for the owner, never retry in a loop.
            logger.warning("Email paused: %s (%s)", code, exc.response.status_code)
        else:
            logger.error("Email send failed: %s %s", exc.response.status_code, exc.response.text)
        return None
    except Exception:
        logger.exception("Email send error")
        return None


def _layout(body_html: str) -> str:
    link = ""
    if APP_URL.startswith("https://"):
        link = (
            f'<p style="margin:22px 0 0;text-align:center">'
            f'<a href="{APP_URL}/" style="display:inline-block;background:#7b2cbf;color:#ffffff;'
            f'text-decoration:none;padding:12px 28px;border-radius:999px;font-weight:bold">'
            f"Visit the Volunteer Party website</a></p>"
        )
    return (
        '<table role="presentation" width="100%" cellpadding="0" cellspacing="0" '
        'style="background:#0d081d;padding:28px 12px"><tr><td align="center">'
        '<table role="presentation" width="560" cellpadding="0" cellspacing="0" '
        'style="background:#160e2e;border-radius:16px;padding:32px;'
        'font-family:Arial,Helvetica,sans-serif;color:#f2ecff;text-align:left">'
        f"{body_html}{link}"
        f'<p style="font-size:12px;color:#9a8fc0;margin:26px 0 0;text-align:center">Sent by '
        f'{escape(EMAIL_FROM_NAME)} — we never ask for your password or card details by email.</p>'
        "</td></tr></table></td></tr></table>"
    )


async def _event_lines_html() -> str:
    try:
        doc = await db.event_settings.find_one({"id": "event-settings"})
    except Exception:
        doc = None
    if not doc:
        return ""
    bits = [escape(str(doc.get(key, ""))) for key in ("event_date", "event_time", "event_location") if doc.get(key)]
    if not bits:
        return ""
    return '<p style="margin:14px 0 0;font-size:14px;color:#cfc4ec">' + " · ".join(bits) + "</p>"


async def send_registration_confirmation(reg: dict) -> str | None:
    """Fixed template — the participant's own registration confirmation (G2: no asks)."""
    first = str(reg.get("first_name", ""))
    subject = "Your Volunteer Party 2025 registration is confirmed"
    song_line = ""
    if reg.get("song_video_id"):
        song_line = (
            '<p style="margin:14px 0 0;font-size:14px;color:#cfc4ec">Your party song request was '
            "received too — our DJ has it on the shortlist.</p>"
        )
    body = (
        f'<h1 style="margin:0 0 10px;font-size:24px;color:#ffffff">See you at the party, {escape(first)}!</h1>'
        '<p style="margin:0;font-size:15px;line-height:1.6">Your registration for Volunteer Party 2025 '
        "is confirmed. Keep your registration ID for check-in:</p>"
        '<p style="margin:14px 0 0"><span style="display:inline-block;background:#241647;'
        'border:1px solid #4c1d95;border-radius:10px;padding:10px 18px;'
        'font-family:Consolas,monospace;font-size:15px;color:#e9d5ff">'
        f"{escape(str(reg.get('id', '')))}</span></p>"
        f"{await _event_lines_html()}{song_line}"
    )
    return await _send_raw(to=str(reg.get("email", "")), subject=subject, html=_layout(body))


async def send_admin_notification(reg: dict, admin_email: str) -> str | None:
    name = f"{reg.get('first_name', '')} {reg.get('last_name', '')}".strip()
    subject = f"New registration — {name}"
    row = lambda label, value: (  # noqa: E731
        f'<tr><td style="padding:6px 12px 6px 0;font-size:13px;color:#9a8fc0;white-space:nowrap">{label}</td>'
        f'<td style="padding:6px 0;font-size:14px;color:#f2ecff">{escape(str(value or "—"))}</td></tr>'
    )
    body = (
        '<h1 style="margin:0 0 10px;font-size:22px;color:#ffffff">New registration received</h1>'
        '<table role="presentation" cellpadding="0" cellspacing="0">'
        f"{row('Name', name)}"
        f"{row('Email', reg.get('email', ''))}"
        f"{row('Phone', reg.get('phone', ''))}"
        f"{row('Volunteer center', reg.get('volunteer_center', ''))}"
        f"{row('Party song', 'Yes' if reg.get('song_video_id') else 'No')}"
        f"{row('Registration ID', reg.get('id', ''))}"
        "</table>"
        '<p style="margin:16px 0 0;font-size:14px;color:#cfc4ec">Manage it in the admin dashboard.</p>'
    )
    return await _send_raw(to=admin_email, subject=subject, html=_layout(body))


def schedule_registration_emails(reg: dict) -> None:
    """Fire-and-forget: participant confirmation + admin notification, logging any failure."""

    async def _run() -> None:
        await send_registration_confirmation(reg)
        admin = await db.admin_users.find_one({}, sort=[("created_at", 1)])
        if admin and admin.get("email"):
            await send_admin_notification(reg, admin["email"])

    async def _safe() -> None:
        try:
            await _run()
        except Exception:
            logger.exception("registration emails failed")

    if EMAIL_ENABLED:
        asyncio.create_task(_safe())
