import { useEffect, useState } from "react";
import { useMutation, useQuery, useQueryClient } from "@tanstack/react-query";
import { toast } from "sonner";
import { CalendarPlus, Megaphone, Pencil, Plus, Save, Trash2 } from "lucide-react";
import { Button } from "@/components/ui/button";
import { Dialog, DialogContent, DialogHeader, DialogTitle } from "@/components/ui/dialog";
import { Input } from "@/components/ui/input";
import { Label } from "@/components/ui/label";
import { Textarea } from "@/components/ui/textarea";
import { apiDelete, apiErrorMessage, apiGet, apiPost, apiPut } from "@/lib/api";
import type { EventBundle, EventSettings, ScheduleItem } from "@/lib/types";

const EMPTY_DRAFT = { title: "", description: "", start_time: "", end_time: "", sort_order: 10 };

export default function EventPanel() {
  const eventQuery = useQuery({
    queryKey: ["event"],
    queryFn: () => apiGet<EventBundle>("/event"),
  });
  const queryClient = useQueryClient();

  const [form, setForm] = useState<EventSettings | null>(null);
  useEffect(() => {
    if (eventQuery.data?.settings) setForm(eventQuery.data.settings);
  }, [eventQuery.data]);

  const set = (patch: Partial<EventSettings>) => setForm((prev) => (prev ? { ...prev, ...patch } : prev));

  const saveSettings = useMutation({
    mutationFn: (payload: Partial<EventSettings>) => apiPut<EventSettings>("/event", payload),
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: ["event"] });
      toast.success("Event information saved.");
    },
    onError: (err) => toast.error(apiErrorMessage(err)),
  });

  const schedule = eventQuery.data?.schedule ?? [];

  const [dialogOpen, setDialogOpen] = useState(false);
  const [editing, setEditing] = useState<ScheduleItem | null>(null);
  const [draft, setDraft] = useState(EMPTY_DRAFT);

  const saveSchedule = useMutation({
    mutationFn: () =>
      editing
        ? apiPut<ScheduleItem>(`/event/schedule/${editing.id}`, draft)
        : apiPost<ScheduleItem>("/event/schedule", draft),
    onSuccess: () => {
      setDialogOpen(false);
      queryClient.invalidateQueries({ queryKey: ["event"] });
      toast.success(editing ? "Schedule item updated." : "Schedule item added.");
    },
    onError: (err) => toast.error(apiErrorMessage(err)),
  });

  const deleteSchedule = useMutation({
    mutationFn: (id: string) => apiDelete(`/event/schedule/${id}`),
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: ["event"] });
      toast.success("Schedule item deleted.");
    },
    onError: (err) => toast.error(apiErrorMessage(err)),
  });

  if (!form) {
    return <div className="h-40 animate-pulse rounded-2xl bg-white/5" data-testid="event-panel-loading" />;
  }

  const announcements = form.announcements ?? [];

  return (
    <div className="space-y-8">
      <div className="glass rounded-2xl p-6" data-testid="event-settings-card">
        <h3 className="font-heading text-lg font-bold text-white">Event information</h3>
        <p className="mt-1 text-xs text-slate-400">
          Every field here updates the public website immediately after saving.
        </p>
        <div className="mt-5 grid gap-4 sm:grid-cols-2">
          <div>
            <Label htmlFor="event-title" className="text-sm text-slate-200">Event title</Label>
            <Input
              id="event-title"
              value={form.event_title}
              onChange={(e) => set({ event_title: e.target.value })}
              data-testid="event-title-input"
              className="mt-1.5 h-10 border-purple-500/25 bg-[#1c143b] text-white"
            />
          </div>
          <div>
            <Label htmlFor="event-organizer" className="text-sm text-slate-200">Organizer</Label>
            <Input
              id="event-organizer"
              value={form.organizer}
              onChange={(e) => set({ organizer: e.target.value })}
              data-testid="event-organizer-input"
              className="mt-1.5 h-10 border-purple-500/25 bg-[#1c143b] text-white"
            />
          </div>
          <div>
            <Label htmlFor="event-date" className="text-sm text-slate-200">Event date</Label>
            <Input
              id="event-date"
              type="date"
              value={form.event_date}
              onChange={(e) => set({ event_date: e.target.value })}
              data-testid="event-date-input"
              className="mt-1.5 h-10 border-purple-500/25 bg-[#1c143b] text-white"
            />
          </div>
          <div>
            <Label htmlFor="event-time" className="text-sm text-slate-200">Start time</Label>
            <Input
              id="event-time"
              type="time"
              value={form.event_time}
              onChange={(e) => set({ event_time: e.target.value })}
              data-testid="event-time-input"
              className="mt-1.5 h-10 border-purple-500/25 bg-[#1c143b] text-white"
            />
          </div>
          <div className="sm:col-span-2">
            <Label htmlFor="event-location" className="text-sm text-slate-200">Location</Label>
            <Input
              id="event-location"
              value={form.event_location}
              onChange={(e) => set({ event_location: e.target.value })}
              data-testid="event-location-input"
              className="mt-1.5 h-10 border-purple-500/25 bg-[#1c143b] text-white"
            />
          </div>
          <div className="sm:col-span-2">
            <Label htmlFor="event-description" className="text-sm text-slate-200">Description (About section lead)</Label>
            <Textarea
              id="event-description"
              value={form.description}
              onChange={(e) => set({ description: e.target.value })}
              data-testid="event-description-input"
              className="mt-1.5 min-h-24 border-purple-500/25 bg-[#1c143b] text-white"
            />
          </div>
          <div>
            <Label htmlFor="contact-email" className="text-sm text-slate-200">Contact email</Label>
            <Input
              id="contact-email"
              type="email"
              value={form.contact_email}
              onChange={(e) => set({ contact_email: e.target.value })}
              data-testid="event-contact-email-input"
              className="mt-1.5 h-10 border-purple-500/25 bg-[#1c143b] text-white"
            />
          </div>
          <div>
            <Label htmlFor="contact-phone" className="text-sm text-slate-200">Contact phone</Label>
            <Input
              id="contact-phone"
              value={form.contact_phone}
              onChange={(e) => set({ contact_phone: e.target.value })}
              data-testid="event-contact-phone-input"
              className="mt-1.5 h-10 border-purple-500/25 bg-[#1c143b] text-white"
            />
          </div>
        </div>
        <Button
          onClick={() => saveSettings.mutate({
            event_title: form.event_title,
            description: form.description,
            event_date: form.event_date,
            event_time: form.event_time,
            event_location: form.event_location,
            organizer: form.organizer,
            contact_email: form.contact_email,
            contact_phone: form.contact_phone,
            announcements: form.announcements,
          })}
          disabled={saveSettings.isPending}
          data-testid="event-settings-save-btn"
          className="mt-5 font-bold tracking-wide"
        >
          <Save className="size-4" /> {saveSettings.isPending ? "Saving…" : "Save event information"}
        </Button>
      </div>

      <div className="glass rounded-2xl p-6" data-testid="announcements-card">
        <h3 className="flex items-center gap-2 font-heading text-lg font-bold text-white">
          <Megaphone className="size-5 text-purple-300" /> Announcements
        </h3>
        <p className="mt-1 text-xs text-slate-400">Shown in the amber banner on the public site.</p>
        <div className="mt-4 space-y-3">
          {announcements.map((text, i) => (
            <div key={i} className="flex gap-2">
              <Input
                value={text}
                onChange={(e) => {
                  const next = [...announcements];
                  next[i] = e.target.value;
                  set({ announcements: next });
                }}
                data-testid={`announcement-input-${i}`}
                className="h-10 border-purple-500/25 bg-[#1c143b] text-white"
              />
              <Button
                type="button"
                variant="ghost"
                size="icon"
                aria-label="Remove announcement"
                data-testid={`announcement-remove-btn-${i}`}
                onClick={() => set({ announcements: announcements.filter((_, idx) => idx !== i) })}
                className="text-slate-400 hover:bg-red-500/15 hover:text-red-300"
              >
                <Trash2 className="size-4" />
              </Button>
            </div>
          ))}
          <Button
            type="button"
            variant="outline"
            onClick={() => set({ announcements: [...announcements, ""] })}
            data-testid="announcement-add-btn"
            className="border-purple-400/40 text-purple-200"
          >
            <Plus className="size-4" /> Add announcement
          </Button>
        </div>
      </div>

      <div className="glass rounded-2xl p-6" data-testid="schedule-card">
        <div className="flex flex-wrap items-center justify-between gap-3">
          <div>
            <h3 className="flex items-center gap-2 font-heading text-lg font-bold text-white">
              <CalendarPlus className="size-5 text-purple-300" /> Event schedule
            </h3>
            <p className="mt-1 text-xs text-slate-400">The public timeline renders in sort order.</p>
          </div>
          <Button
            onClick={() => {
              setEditing(null);
              setDraft({ ...EMPTY_DRAFT, sort_order: (schedule.length + 1) * 10 });
              setDialogOpen(true);
            }}
            data-testid="schedule-add-btn"
            className="font-bold tracking-wide"
          >
            <Plus className="size-4" /> Add item
          </Button>
        </div>
        {schedule.length === 0 ? (
          <p className="mt-4 text-sm text-slate-400" data-testid="schedule-admin-empty">No schedule items yet.</p>
        ) : (
          <ul className="mt-5 divide-y divide-white/5">
            {schedule.map((item) => (
              <li key={item.id} className="flex items-center justify-between gap-3 py-3" data-testid={`schedule-row-${item.id}`}>
                <div className="min-w-0">
                  <p className="font-mono text-xs text-purple-300">
                    {item.start_time}
                    {item.end_time ? ` – ${item.end_time}` : ""} · order {item.sort_order}
                  </p>
                  <p className="truncate text-sm font-semibold text-white">{item.title}</p>
                  {item.description && <p className="truncate text-xs text-slate-400">{item.description}</p>}
                </div>
                <div className="flex shrink-0 gap-1.5">
                  <Button
                    size="icon-sm"
                    variant="ghost"
                    aria-label="Edit schedule item"
                    data-testid={`schedule-edit-btn-${item.id}`}
                    onClick={() => {
                      setEditing(item);
                      setDraft({
                        title: item.title,
                        description: item.description,
                        start_time: item.start_time,
                        end_time: item.end_time,
                        sort_order: item.sort_order,
                      });
                      setDialogOpen(true);
                    }}
                  >
                    <Pencil className="size-4" />
                  </Button>
                  <Button
                    size="icon-sm"
                    variant="ghost"
                    aria-label="Delete schedule item"
                    data-testid={`schedule-delete-btn-${item.id}`}
                    disabled={deleteSchedule.isPending}
                    onClick={() => {
                      if (window.confirm(`Delete "${item.title}" from the schedule?`)) deleteSchedule.mutate(item.id);
                    }}
                    className="text-slate-400 hover:bg-red-500/15 hover:text-red-300"
                  >
                    <Trash2 className="size-4" />
                  </Button>
                </div>
              </li>
            ))}
          </ul>
        )}
      </div>

      <Dialog open={dialogOpen} onOpenChange={setDialogOpen}>
        <DialogContent className="glass-strong max-w-md rounded-2xl" data-testid="schedule-dialog">
          <DialogHeader>
            <DialogTitle className="font-heading text-lg font-bold text-white">
              {editing ? "Edit schedule item" : "New schedule item"}
            </DialogTitle>
          </DialogHeader>
          <div className="space-y-3">
            <div>
              <Label htmlFor="schedule-title" className="text-sm text-slate-200">Title *</Label>
              <Input
                id="schedule-title"
                value={draft.title}
                onChange={(e) => setDraft({ ...draft, title: e.target.value })}
                data-testid="schedule-title-input"
                className="mt-1.5 h-10 border-purple-500/25 bg-[#1c143b] text-white"
              />
            </div>
            <div>
              <Label htmlFor="schedule-description" className="text-sm text-slate-200">Description</Label>
              <Textarea
                id="schedule-description"
                value={draft.description}
                onChange={(e) => setDraft({ ...draft, description: e.target.value })}
                data-testid="schedule-description-input"
                className="mt-1.5 min-h-16 border-purple-500/25 bg-[#1c143b] text-white"
              />
            </div>
            <div className="grid grid-cols-3 gap-3">
              <div>
                <Label htmlFor="schedule-start" className="text-sm text-slate-200">Start</Label>
                <Input
                  id="schedule-start"
                  type="time"
                  value={draft.start_time}
                  onChange={(e) => setDraft({ ...draft, start_time: e.target.value })}
                  data-testid="schedule-start-input"
                  className="mt-1.5 h-10 border-purple-500/25 bg-[#1c143b] text-white"
                />
              </div>
              <div>
                <Label htmlFor="schedule-end" className="text-sm text-slate-200">End</Label>
                <Input
                  id="schedule-end"
                  type="time"
                  value={draft.end_time}
                  onChange={(e) => setDraft({ ...draft, end_time: e.target.value })}
                  data-testid="schedule-end-input"
                  className="mt-1.5 h-10 border-purple-500/25 bg-[#1c143b] text-white"
                />
              </div>
              <div>
                <Label htmlFor="schedule-order" className="text-sm text-slate-200">Order</Label>
                <Input
                  id="schedule-order"
                  type="number"
                  value={draft.sort_order}
                  onChange={(e) => setDraft({ ...draft, sort_order: Number(e.target.value) || 0 })}
                  data-testid="schedule-order-input"
                  className="mt-1.5 h-10 border-purple-500/25 bg-[#1c143b] text-white"
                />
              </div>
            </div>
            <Button
              onClick={() => saveSchedule.mutate()}
              disabled={saveSchedule.isPending || draft.title.trim().length < 2}
              data-testid="schedule-save-btn"
              className="w-full font-bold tracking-wide"
            >
              <Save className="size-4" /> {saveSchedule.isPending ? "Saving…" : "Save schedule item"}
            </Button>
          </div>
        </DialogContent>
      </Dialog>
    </div>
  );
}
