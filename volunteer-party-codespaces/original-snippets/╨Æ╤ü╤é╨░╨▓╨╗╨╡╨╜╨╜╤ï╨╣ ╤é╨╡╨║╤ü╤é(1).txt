function Countdown({ date, time, serverNow }: { date: string; time: string; serverNow?: string }) {
  const [nowMs, setNowMs] = useState(() => (serverNow ? Date.parse(serverNow) : Date.now()));

  // Anchor "now" to the server clock, then tick locally (drift-free, no client date math).
  useEffect(() => {
    if (serverNow) setNowMs(Date.parse(serverNow));
  }, [serverNow]);

  useEffect(() => {
    const id = setInterval(() => setNowMs((m) => m + 1000), 1000);
    return () => clearInterval(id);
  }, []);

  if (!date) return null;
  const target = Date.parse(`${date}T${time || "19:00"}:00`);
  if (Number.isNaN(target)) return null;
  const diff = target - nowMs;
  if (diff <= 0) {
    return (
      <div data-testid="hero-countdown-timer" className="glass rounded-xl px-6 py-4 font-heading text-xl font-bold text-purple-200">
        It is party time!
      </div>
    );
  }
  const cells = [
    { label: "Days", value: Math.floor(diff / 86_400_000) },
    { label: "Hours", value: Math.floor((diff % 86_400_000) / 3_600_000) },
    { label: "Minutes", value: Math.floor((diff % 3_600_000) / 60_000) },
    { label: "Seconds", value: Math.floor((diff % 60_000) / 1000) },
  ];
  return (
    <div data-testid="hero-countdown-timer" className="flex gap-2 sm:gap-3">
      {cells.map((cell) => (
        <div key={cell.label} className="glass w-16 rounded-xl px-2 py-2 text-center sm:w-20 sm:py-3">
          <div className="font-mono text-2xl font-bold text-white sm:text-3xl">
            {String(cell.value).padStart(2, "0")}
          </div>
          <div className="text-[10px] font-semibold uppercase tracking-[0.2em] text-purple-300">{cell.label}</div>
        </div>
      ))}
    </div>
  );
}
